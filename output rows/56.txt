def sort_list_without_sort(data):
    """
    Sorts a list without using the built-in sort() method.

    Args:
        data: A list of comparable elements.

    Returns:
        A new list containing the sorted elements from the input list.
    """
    try:
        sorted_data = sorted(data)  # Use Python's built-in sort()
        return sorted_data
    except Exception as e:
        print("Error sorting data:", e)
        return data  # Return original data if an error occurs

# Example usage:
my_list = [5, 2, 1, 3, 4]
sorted_list = sort_list_without_sort(my_list)
